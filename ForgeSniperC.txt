-- game modules need the game's thread identity
local canSetIdentity = type(setthreadidentity) == "function"
local execIdentity = 7
if type(getthreadidentity) == "function" then
    local ok, id = pcall(getthreadidentity)
    if ok and type(id) == "number" then execIdentity = id end
end
local function gcall(fn, ...)
    if type(fn) ~= "function" then return false, "not a function" end
    if canSetIdentity then pcall(setthreadidentity, 2) end
    local r = table.pack(pcall(fn, ...))
    if canSetIdentity then pcall(setthreadidentity, execIdentity) end
    return table.unpack(r, 1, r.n)
end
local function gval(fn, ...)
    local ok, a = gcall(fn, ...)
    if ok then return a end
    return nil
end

local function notify(title, text)
    pcall(function()
        StarterGui:SetCore("SendNotification", { Title = title, Text = text, Duration = 5 })
    end)
end

local function req(path)
    local ok, mod = gcall(function()
        local node = RS
        for seg in string.gmatch(path, "[^%.]+") do
            node = node:WaitForChild(seg, 15)
            if not node then error("missing " .. path) end
        end
        return require(node)
    end)
    return ok and mod or nil
end

local function isHuge(id)
    return string.find(string.lower(tostring(id)), "huge", 1, true) ~= nil

local function inEvent()
    if type(Instancing) ~= "table" or type(Instancing.GetInstanceID) ~= "function" then return true end

local function rootPart()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart") or nil
end
local function atPad()
    local r = rootPart()
    return r ~= nil and (r.Position - FORGE_PAD).Magnitude <= PAD_RADIUS
end
local function gotoPad()
    local c, r = LocalPlayer.Character, rootPart()
    if not (c and r) then return false end
    pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
    return (pcall(function() c:PivotTo(CFrame.new(FORGE_PAD + Vector3.new(0, 3, 0))) end))
end
local function walkToPad()
    local c, r = LocalPlayer.Character, rootPart()
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    if not (c and r and hum) then return false end
    local t0, lastMove, lastPos = os.clock(), os.clock(), r.Position
    while alive and os.clock() - t0 < 25 do
        r = rootPart()
        if not r then return false end
        local flat = Vector3.new(r.Position.X - FORGE_PAD.X, 0, r.Position.Z - FORGE_PAD.Z).Magnitude
        if flat <= 5 then return true end
        hum:MoveTo(FORGE_PAD)
        -- hop over small obstacles if we stop making progress
        if (r.Position - lastPos).Magnitude > 1 then
            lastPos, lastMove = r.Position, os.clock()
        elseif os.clock() - lastMove > 1.2 then
            hum.Jump = true
            lastMove = os.clock()
        end
        task.wait(0.1)
    end
    return atPad()
end
local function goToPad()
    if S.travel == "Walk" then
        return walkToPad()
    end
    local ok = gotoPad()
    task.wait(0.25)
    return ok
end

local function scanTargets(st)
    local out = {}
    local targets = type(ForgeUtil.TARGETS) == "table" and ForgeUtil.TARGETS or {}
    for id, t in pairs(targets) do
        local v = type(st) == "table" and st[id] or nil
        local info = { id = id, open = false }
        if type(v) == "table" and type(v.left) == "number" then
            info.left = v.left
        end
        if type(v) == "table" and type(v.dial) == "number" then
            info.dial = v.dial
        end
        if type(v) == "table" and flag(t.Enabled) == true and v.open and type(v.dial) == "number" and type(v.left) == "number" then
            local hasLimit = type(v.limit) == "number" and v.limit >= 0
            info.dial = v.dial
            if v.left > 0 and not (hasLimit and (type(v.used) ~= "number" or v.used >= v.limit)) then
                info.open = true
            end
        end
        out[id] = info
    end
    return out
end

-- pets
local function equippedSet()
    local set = {}
    if type(PetCmds) == "table" and type(PetCmds.GetEquippedItems) == "function" then
        local ok, items = gcall(PetCmds.GetEquippedItems)
        if ok and type(items) == "table" then
            for _, it in pairs(items) do
                if type(it) == "string" then
                    set[it] = true
                else
                    local okU, uid = pcall(function() return it:GetUID() end)
                    if okU and uid then set[uid] = true end
                end
            end
        end
    end
    return set
end

local function collectPets(forDisplay)
    local out = {}
    local inputs = gval(ForgeUtil.Inputs, Forge.flag)
    if type(inputs) ~= "table" then return out end
    local ok, all = gcall(function() return Items.Pet:All() end)
    if not (ok and type(all) == "table") then return out end
    local equipped = (not forDisplay and S.skipEquipped) and equippedSet() or {}
    for uid, item in pairs(all) do
        local okI, id, amount, variant = pcall(function()
            return item:GetId(), item:GetAmount(), (item:IsGolden() or item:IsRainbow() or item:IsShiny()) and 1 or 0
        end)
        if okI and inputs[id] and (amount or 0) > 0 and not equipped[uid] and (forDisplay or S.selected[id]) then
            out[#out + 1] = { uid = uid, id = id, amount = amount, power = inputs[id], variant = variant, huge = isHuge(id) }
        end
    end
    table.sort(out, function(a, b)
        if a.power ~= b.power then return a.power < b.power end
        if a.variant ~= b.variant then return a.variant < b.variant end
        if a.amount ~= b.amount then return a.amount > b.amount end
        return tostring(a.uid) < tostring(b.uid)
    end)
    return out
end

local function tryAdd(c, amount, ctx)
    if amount <= 0 then return "skip" end
    ctx.counts[c.id] = (ctx.counts[c.id] or 0) + amount
    local ok, tbl = gcall(ForgeUtil.Table, Forge.flag, ctx.counts, ctx.shards, ctx.dials, ctx.closed)
    local bad = not (ok and type(tbl) == "table")
    if bad or tbl.overflow or #tbl.rows == 0 then
        ctx.counts[c.id] -= amount
        if ctx.counts[c.id] <= 0 then ctx.counts[c.id] = nil end
        if bad then return "error" end
        return #tbl.rows == 0 and "empty" or "overflow"
    end
    ctx.pets[c.uid] = (ctx.pets[c.uid] or 0) + amount
    ctx.used += amount
    if c.huge then ctx.huges += amount else ctx.normals += amount end
    ctx.best = tbl
    return tbl.capped and "capped" or "ok"
end

local function buildPlan(cands, targets)
    local dials, closed = {}, {}
    for id, info in pairs(targets) do
        if info.open then dials[id] = info.dial else closed[id] = true end
    end
    local shards = 0
    if S.useShards and type(Forge.Shards) == "function" and type(ForgeUtil.ShardCap) == "function" then
        shards = math.min(gval(Forge.Shards) or 0, gval(ForgeUtil.ShardCap, Forge.flag) or 0)
    end
    local ctx = { counts = {}, pets = {}, used = 0, huges = 0, normals = 0, shards = shards, dials = dials, closed = closed }

    -- first make sure the minimum huge / normal amounts are in
    for _, c in ipairs(cands) do
        local need = c.huge and (S.minHuge - ctx.huges) or (S.minNormal - ctx.normals)
        if need > 0 then
            local r = tryAdd(c, math.min(c.amount, need), ctx)
            if r == "error" then return nil, "forge table error" end
        end
    end
    -- then everything else that was selected
    for _, c in ipairs(cands) do
        local r = tryAdd(c, c.amount - (ctx.pets[c.uid] or 0), ctx)
        if r == "error" then return nil, "forge table error" end
        if r == "capped" then break end
    end

    if ctx.used == 0 or not ctx.best then return nil, "nothing to win with these pets (targets closed or pets too strong)" end
    if ctx.huges < S.minHuge or ctx.normals < S.minNormal then
        return nil, string.format("plan only fits %d huge / %d normal, need %d / %d", ctx.huges, ctx.normals, S.minHuge, S.minNormal)
    end
    return { pets = ctx.pets, shards = shards, total = ctx.best.total, used = ctx.used }
end

-- turns the forge table rows into { name, pct } entries (names of the actual pets)
local NAME_KEYS = { "Name", "name", "DisplayName", "Result", "result", "PetId", "Pet", "pet", "Id", "id", "Target", "target" }
local CHANCE_KEYS = { "chance", "Chance", "p", "P", "prob", "Prob", "probability", "Probability", "odds", "Odds", "pct", "Pct" }
local fieldsLogged = false
local function pickField(row, keys, kind)
    for _, k in ipairs(keys) do
        local v = row[k]
        if type(v) == kind then return v end
    end
    return nil
end
local function readRows(tbl)
    local out = { named = false }
    if type(tbl.rows) ~= "table" then return out end
    if not fieldsLogged and type(tbl.rows[1]) == "table" then
        fieldsLogged = true
        local keys = {}
        for k, v in pairs(tbl.rows[1]) do keys[#keys + 1] = tostring(k) .. "=" .. typeof(v) end
        table.sort(keys)
        S.rowFields = table.concat(keys, ", ")
        print("[ForgeSniper] forge row fields: " .. S.rowFields)
    end
    local sum = 0
    for _, r in ipairs(tbl.rows) do
        if type(r) == "table" then
            local name, chance = pickField(r, NAME_KEYS, "string"), pickField(r, CHANCE_KEYS, "number")
            if name and chance then
                out[#out + 1] = { name = name, pct = chance }
                sum += chance
            end
        end
    end
    if #out == 0 then return out end
    -- rows may hold fractions (0-1) or percents, compare with the table total
    local total = tonumber(tbl.total) or 0
    local scale = 100
    if math.abs(sum - total * 100) < math.max(1, total * 5) and total > 0 and sum > 1.5 then scale = 1 end
    for _, e in ipairs(out) do e.pct *= scale end
    table.sort(out, function(a, b) return a.pct > b.pct end)
    out.named = true
    return out
end

-- win chance for one target with all ticked pets, as if only that target were open
S.chances = {}
local function updateChances()
    local targets = S.targets
    local cands = collectPets(false)
    local shards = 0
    if S.useShards and type(Forge.Shards) == "function" and type(ForgeUtil.ShardCap) == "function" then
        shards = math.min(gval(Forge.Shards) or 0, gval(ForgeUtil.ShardCap, Forge.flag) or 0)
    end
    for id, info in pairs(targets) do
        local res
        if #cands == 0 then
            res = { text = "tick pets to see" }
        elseif type(info.dial) ~= "number" then
            res = { text = "n/a" }
        else
            local closed = {}
            for other in pairs(targets) do
                if other ~= id then closed[other] = true end
            end
            local ctx = { counts = {}, pets = {}, used = 0, huges = 0, normals = 0, shards = shards, dials = { [id] = info.dial }, closed = closed }
            for _, c in ipairs(cands) do
                local r = tryAdd(c, c.amount, ctx)
                if r == "error" or r == "capped" then break end
            end
            if ctx.best then
                res = { pct = ctx.best.total * 100, text = string.format("%.1f%%", ctx.best.total * 100), rows = readRows(ctx.best) }
            else
                res = { text = "0% (pets too weak)" }
            end
        end
        S.chances[id] = res
    end
end

local function attempt()
    local st = gval(Forge.State)
    if type(st) ~= "table" then st = gval(Forge.Fetch) end
    if type(st) ~= "table" then S.note = "forge state not loaded" return false end
    local targets = scanTargets(st)
    local cands = collectPets(false)
    if #cands == 0 then S.note = "none of the ticked pets are in your inventory" return false end

    local hugeHave, normalHave = 0, 0
    for _, c in ipairs(cands) do
        if c.huge then hugeHave += c.amount else normalHave += c.amount end
    end
    if hugeHave < S.minHuge or normalHave < S.minNormal then
        S.note = string.format("need %d huge / %d normal, have %d / %d", S.minHuge, S.minNormal, hugeHave, normalHave)
        return false
    end

    local plan, why = buildPlan(cands, targets)
    if not plan then S.note = tostring(why) return false end

    if S.autoTp and inEvent() and not atPad() then
        S.note = S.travel == "Walk" and "walking to the forge" or "teleporting to the forge"
        goToPad()
    end

    local ok, res = gcall(Forge.Roll, plan.pets, plan.shards)
    S.rolls += 1
    if ok and type(res) == "table" and res.ok then
        if res.hit then
            S.wins += 1
            local what = tostring(res.result or res.target)
            S.note = string.format("WON %s (%d pets, %.1f%%)", what, plan.used, plan.total * 100)
            notify("Forge Sniper", "You forged " .. what .. "!")
        else
            S.note = string.format("missed (%.1f%%, %d pets used)", plan.total * 100, plan.used)
        end
        return true
    end
    S.note = "roll refused: " .. tostring(type(res) == "table" and res.msg or res)
    return false
end

local function arm(on)
    S.armed = on
    table.clear(last)
    table.clear(pending)
    if on then
        S.note = "armed, watching the forge"
        if S.autoTp and inEvent() and not atPad() then task.spawn(goToPad) end
    else
        S.note = "stopped"
    end
end

track(RunService.Heartbeat:Connect(function()
    if not (alive and S.armed) then return end
    local st = gval(Forge.State)
    if type(st) ~= "table" then return end
    local targets = scanTargets(st)
    S.targets = targets
    local fire = false
    for id, info in pairs(targets) do
        local prev = last[id]
        if info.open and S.trigger[id] then
            if not prev or not prev.open or (info.left or 0) > (prev.left or 0) then pending[id] = true end
        else
            pending[id] = nil
        end
        last[id] = { open = info.open, left = info.left }
        if pending[id] then fire = true end
    end
    if fire and not busy and os.clock() >= nextTry then
        busy = true
        task.spawn(function()
            local ok, fired = pcall(attempt)
            if ok and fired then
                nextTry = os.clock() + 1
                if not S.keepRolling then table.clear(pending) end
            else
                if not ok then S.note = "error: " .. tostring(fired) end
                nextTry = os.clock() + 0.25
            end
            busy = false
        end)
    end
end))

    while alive do
        if S.armed and type(Forge.Fetch) == "function" then gcall(Forge.Fetch) end
        task.wait(0.4)
    end
end)

local function segmented(parent, text, options, default, cb)
    local wrap = new("Frame", { BackgroundColor3 = C.item, Size = UDim2.new(1, 0, 0, 68), LayoutOrder = nxt(parent) }, parent)
    round(wrap, 8)
    new("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(10, 4), Size = UDim2.new(1, -20, 0, 22),
        Text = text, Font = Enum.Font.Gotham, TextSize = 14, TextColor3 = C.text, TextXAlignment = Enum.TextXAlignment.Left,
    }, wrap)
    local track_ = new("Frame", { Position = UDim2.fromOffset(8, 28), Size = UDim2.new(1, -16, 0, 32), BackgroundColor3 = C.bg }, wrap)
    round(track_, 8)
    new("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }, track_)
    new("UIPadding", { PaddingTop = UDim.new(0, 3), PaddingBottom = UDim.new(0, 3), PaddingLeft = UDim.new(0, 3), PaddingRight = UDim.new(0, 3) }, track_)
    local n = #options
    local btns = {}
    local current = default
    local function paint()
        for name, b in pairs(btns) do
            b.BackgroundColor3 = name == current and C.accent or C.bg
            b.TextColor3 = name == current and Color3.new(1, 1, 1) or C.dim
        end
    end
    for i, name in ipairs(options) do
        local b = new("TextButton", {
            Size = UDim2.new(1 / n, -4 * (n - 1) / n, 1, 0), Text = name, Font = Enum.Font.GothamBold, TextSize = 13,
            BackgroundColor3 = C.bg, TextColor3 = C.dim, AutoButtonColor = false, LayoutOrder = i,
        }, track_)
        round(b, 6)
        b.Activated:Connect(function()
            current = name
            paint()
            cb(name)
        end)
        btns[name] = b
    end
    paint()
end

function S.refresh()
    local list = collectPets(true)
    local groups, ids = {}, {}
    for _, c in ipairs(list) do
        local g = groups[c.id]
        if not g then
            g = { id = c.id, amount = 0, huge = c.huge }
            groups[c.id] = g
            ids[#ids + 1] = c.id
        end
        g.amount += c.amount
    end
    table.sort(ids, function(a, b)
        local ha, hb = groups[a].huge, groups[b].huge
        if ha ~= hb then return ha end
        return tostring(a) < tostring(b)
    end)
    for i, id in ipairs(ids) do
        local g = groups[id]
        if not S.rows[id] then S.rows[id] = petRow(id, g.huge) end
        local r = S.rows[id]
        r.setCount(g.amount)
        r.frame.LayoutOrder = i
        r.frame.Visible = true
    end
    for id, r in pairs(S.rows) do
        if not groups[id] then r.frame.Visible = false end
    end
    lblEmpty.Visible = #ids == 0
end

local function stop()
    if not alive then return end
    alive = false
    S.armed = false
    for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    conns = {}
    pcall(function() gui:Destroy() end)
    genv.ForgeSniper = nil
end
btnClose.Activated:Connect(stop)

-- ui refresh
task.spawn(function()
    local tick = 0
    while alive do
        if not S.armed then
            local st = gval(Forge.State)
            if type(st) == "table" then S.targets = scanTargets(st) end
        end
        if tick % 12 == 0 then pcall(S.refresh) end
        if tick % 4 == 0 and not busy then pcall(updateChances) end
        tick += 1

        lblState.Text = S.armed and "ARMED, waiting for stock" or "Sniper off"
        lblState.TextColor3 = S.armed and C.good or C.text
        dot.BackgroundColor3 = S.armed and C.good or C.off
        armBtn.Text = S.armed and "Stop Sniper" or "Start Sniper"
        armBtn.BackgroundColor3 = S.armed and C.bad or C.accent

        local ids = {}
        for id in pairs(S.targets) do ids[#ids + 1] = id end
        table.sort(ids, function(a, b) return tostring(a) < tostring(b) end)
        local lines = {}
        for _, id in ipairs(ids) do
            local info = S.targets[id]
            local txt
            if info.open then txt = tostring(info.left) .. " left"
            elseif info.left == 0 then txt = "sold out"
            else txt = "closed" end
            local ch = S.chances[id]
            local head = (S.trigger[id] and "● " or "○ ") .. tostring(id) .. ": " .. txt
            if ch and ch.rows and ch.rows.named then
                if #ch.rows > 1 then head = head .. "  (total " .. ch.text .. ")" end
                lines[#lines + 1] = head
                for i, e in ipairs(ch.rows) do
                    if i > 20 then lines[#lines + 1] = "     ..." break end
                    lines[#lines + 1] = string.format("     %s: %.1f%%", e.name, e.pct)
                end
            else
                lines[#lines + 1] = head .. "\n     chance: " .. (ch and ch.text or "...")
                if ch and ch.rows and S.rowFields and ch.pct then
                    lines[#lines + 1] = "     (pet names not found, row fields: " .. S.rowFields .. ")"
                end
            end
        end
        lblStock.Text = #lines > 0 and table.concat(lines, "\n") or "Stock: -"
        lblNote.Text = S.note
        lblStats.Text = string.format("Rolls: %d | Wins: %d", S.rolls, S.wins)
        task.wait(0.25)
    end
end)
