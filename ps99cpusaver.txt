local G=getgenv()
if G.CS then for _,c in G.CS do pcall(function()c:Disconnect()end)end end
local cs={}
G.CS=cs
local function C(s,f)cs[#cs+1]=s:Connect(f)end

if setfpscap and G.FpsCap then setfpscap(G.FpsCap)end

pcall(function()
	local r=settings().Rendering
	r.QualityLevel=Enum.QualityLevel.Level01
	r.MeshPartDetailLevel=Enum.MeshPartDetailLevel.DistanceBased
end)

if not game:IsLoaded()then game.Loaded:Wait()end

local P=game:GetService("Players")
local RS=game:GetService("RunService")
local L=game:GetService("Lighting")
local SG=game:GetService("StarterGui")
local lp=P.LocalPlayer
local pg=lp:WaitForChild("PlayerGui")
local ws=workspace

task.wait(5)

if G.FPSUI==true then
	local cg=game:GetService("CoreGui")
	local o=cg:FindFirstChild("FPSDisplay")
	if o then o:Destroy()end
	local g=Instance.new("ScreenGui")
	g.Name="FPSDisplay"
	g.IgnoreGuiInset=true
	g.DisplayOrder=999
	g.ResetOnSpawn=false
	g.Parent=cg
	local t=Instance.new("TextLabel")
	t.Size=UDim2.new(0,100,0,35)
	t.Position=UDim2.new(0,45,0,90)
	t.BackgroundTransparency=1
	t.TextColor3=Color3.fromRGB(0,255,100)
	t.Font=Enum.Font.GothamBold
	t.TextSize=26
	t.TextXAlignment=Enum.TextXAlignment.Left
	t.Parent=g
	local s=Instance.new("UIStroke")
	s.Color=Color3.new()
	s.Thickness=1.5
	s.Transparency=.3
	s.Parent=t
	local n,t0=0,os.clock()
	C(RS.Heartbeat,function()
		n+=1
		local d=os.clock()-t0
		if d<1 then return end
		t.Text="FPS: "..math.floor(n/d)
		n=0
		t0=os.clock()
	end)
end

local ch=lp.Character or lp.CharacterAdded:Wait()
ch:WaitForChild("HumanoidRootPart")

local pt=Instance.new("Part")
pt.Size=Vector3.new(15,1,15)
pt.Anchored=true
pt.CanCollide=true
pt.Transparency=1
pt.Parent=ws

local function sc(i)
	if i:IsA("LuaSourceContainer")then return true end
	return(i:FindFirstAncestorWhichIsA("LuaSourceContainer")or i:FindFirstAncestor("ClientModule"))~=nil
end

local function fx(d)
	if d:IsA("BillboardGui")then
		d.Enabled=false
	elseif d:IsA("ParticleEmitter")or d:IsA("Trail")or d:IsA("Smoke")or d:IsA("Fire")or d:IsA("Beam")or d:IsA("Sparkles")or d:IsA("Light")or d:IsA("Sky")or d:IsA("Atmosphere")then
		d:Destroy()
	elseif d:IsA("PostEffect")then
		d.Enabled=false
	end
end

local function hd(d)
	if d:IsA("BasePart")then
		d.Transparency=1
		d.CastShadow=false
		if d:IsA("MeshPart")then d.TextureID=""end
	elseif d:IsA("Decal")or d:IsA("Texture")then
		d.Transparency=1
	end
end

local function pr(d)
	fx(d)
	if not d:IsDescendantOf(ws)then return end
	local c=lp.Character
	if c and d:IsDescendantOf(c)then return end
	if sc(d)then return end
	hd(d)
end

local q,qh,qt={},1,0
local function push(d)qt+=1 q[qt]=d end

C(RS.Heartbeat,function()
	local t=os.clock()
	while qh<=qt and os.clock()-t<.004 do
		local d=q[qh]
		q[qh]=nil
		qh+=1
		pcall(pr,d)
	end
	if qh>qt then qh,qt=1,0 end
end)

C(game.DescendantAdded,push)
for _,d in game:GetDescendants()do push(d)end

task.spawn(function()
	if not getnilinstances then return end
	for i,d in getnilinstances()do
		pcall(function()
			if d:IsA("BasePart")then d.Transparency=1 end
			for _,x in d:GetDescendants()do
				if x:IsA("BasePart")then x.Transparency=1 end
			end
		end)
		if i%100==0 then task.wait()end
	end
end)

local function cv(c)
	if not c or c==lp.Character then return end
	local h=c:FindFirstChildOfClass("Humanoid")or c:WaitForChild("Humanoid",5)
	if h then
		h.DisplayDistanceType=Enum.HumanoidDisplayDistanceType.None
		h.HealthDisplayType=Enum.HumanoidHealthDisplayType.AlwaysOff
		local a=h:FindFirstChildOfClass("Animator")
		if a then a:Destroy()end
		Instance.new("Animator").Parent=h
	end
	for _,d in c:GetDescendants()do
		if d:IsA("BasePart")or d:IsA("Decal")then
			d.Transparency=1
		elseif d:IsA("BillboardGui")then
			d.Enabled=false
		elseif d:IsA("ParticleEmitter")or d:IsA("Trail")or d:IsA("Beam")then
			d:Destroy()
		end
	end
end

local function hp(p)
	if p==lp then return end
	C(p.CharacterAdded,function(c)task.spawn(pcall,cv,c)end)
	if p.Character then task.spawn(pcall,cv,p.Character)end
end

for _,p in P:GetPlayers()do hp(p)end
C(P.PlayerAdded,hp)

if G.DeleteUI==true then
	pcall(function()SG:SetCoreGuiEnabled(Enum.CoreGuiType.All,false)end)
	local wl={CoordinateTrackerGui=true,SimpleAutoClickGui=true,ScanGUI=true}
	local function ui(d)
		if d:IsA("ScreenGui")then
			if not wl[d.Name]then d.Enabled=false end
		elseif d:IsA("BillboardGui")or d:IsA("SurfaceGui")then
			d.Enabled=false
		elseif d:IsA("GuiObject")then
			local s=d:FindFirstAncestorOfClass("ScreenGui")
			if not(s and wl[s.Name])then d.Visible=false end
		end
	end
	for _,d in pg:GetDescendants()do pcall(ui,d)end
	C(pg.DescendantAdded,function(d)pcall(ui,d)end)
end

local cam=ws.CurrentCamera
if cam then
	local function cc(c)
		if c:IsA("Model")or c:IsA("BasePart")then c:Destroy()end
	end
	for _,c in cam:GetChildren()do cc(c)end
	C(cam.ChildAdded,cc)
end

pcall(function()
	L.GlobalShadows=false
	local t=ws.Terrain
	t.WaterWaveSize=0
	t.WaterWaveSpeed=0
	t.WaterReflectance=0
end)

if G.NoRender==true then
	pcall(function()RS:Set3dRenderingEnabled(false)end)
end

local id=os.clock()
G.CSID=id
task.spawn(function()
	while task.wait(15)and G.CSID==id do
		pcall(collectgarbage,"collect")
	end
end)
