---------------------------------------------------------------- number helpers
local function parseNum(s)
	if type(s) ~= "string" then return tonumber(s) end
	s = s:gsub("[%s,]", "")
	if s == "" then return nil end
	local num, suf = s:match("^(%d+%.?%d*)([kmbtKMBT]?)$")
	if not num then return tonumber(s) end
	local mult = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }
	return tonumber(num) * (mult[suf:lower()] or 1)
end

local UNITS = { { 1e12, "t" }, { 1e9, "b" }, { 1e6, "m" }, { 1e3, "k" } }
local function fmt(n)
	n = tonumber(n) or 0
	local a = math.abs(n)
	for _, u in ipairs(UNITS) do
		if a >= u[1] then
			local s = string.format("%.2f", n / u[1]):gsub("%.?0+$", "")
			return s .. u[2]
		end
	end
	return tostring(math.floor(n))
end

---------------------------------------------------------------- tiered items (Coins IV, Strong Pets V ...)
local ROMAN = { "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX", "X" }
local ROMAN_VALUE = {}
for i, r in ipairs(ROMAN) do ROMAN_VALUE[r] = i end
local TIER_MAX = { Enchant = 5, Potion = 10 }

-- "Coins IV" -> "Coins", 4.  Names without a numeral come back unchanged.
local function splitTier(name)
	local base, numeral = name:match("^(.-)%s+([IVXivx]+)$")
	if base and base ~= "" and ROMAN_VALUE[numeral:upper()] then return base, ROMAN_VALUE[numeral:upper()] end
	return name, nil
end

local function expandTiers(class, list)
	local out, max = {}, TIER_MAX[class]
	for _, n in ipairs(list) do
		if max then
			for t = 1, max do out[#out + 1] = { label = n .. " " .. ROMAN[t], name = n, tier = t } end
		else
			out[#out + 1] = { label = n, name = n }
		end
	end
	return out
end

local function itemTitle(r)
	local name = tostring(r.Item)
	if TIER_MAX[r.Class] and r.Tier and ROMAN[r.Tier] then name = name .. " " .. ROMAN[r.Tier] end
	return string.format("%s  ·  %s", name, tostring(r.Class))
end

-- names are compared without case, spaces or punctuation, and a rule for "Wisp Griffin" also
-- catches the Huge / Titanic / Gargantuan version of it
local function norm(s) return (tostring(s):lower():gsub("[^%w]", "")) end
local function stripSize(name)
	local out = name:gsub("^[Hh]uge%s+", ""):gsub("^[Tt]itanic%s+", ""):gsub("^[Gg]argantuan%s+", "")
	return out
end

local function hasActiveRules(pname)
	local prof = cfg.rules[pname]
	local function live(r) return r.Enabled ~= false and (r.MaxPrice ~= nil or r.Profit ~= nil) end
	for _, r in ipairs(prof.items) do if live(r) then return true end end
	for _, r in ipairs(prof.custom) do if live(r) then return true end end
	for _, key in ipairs(EXTRA_KEYS) do
		if prof.extras[key] and live(prof.extras[key]) then return true end
	end
	return false
end

---------------------------------------------------------------- rule matching
local index = { hop = {}, afk = {} }
local cfgVersion = 0
local blCache = setmetatable({}, { __mode = "k" })

local function splitList(s)
	local t = {}
	for w in string.gmatch(s or "", "[^,]+") do
		w = (w:gsub("^%s+", ""))
		w = (w:gsub("%s+$", ""))
		if #w > 0 then t[w:lower()] = true end
	end
	return t
end

local EXTRA_MATCH = {
	Gargantuan = function(c) return c.class == "Pet" and c.garg end,
	Titanic = function(c) return c.class == "Pet" and c.titanic end,
	Huge = function(c) return c.class == "Pet" and c.huge end,
	Exclusive = function(c) return c.class == "Pet" and c.rarity == "Exclusive" end,
	Any = function() return true end,
}

local function rebuild()
	cfgVersion += 1
	for pname, prof in pairs(cfg.rules) do
		local byClassName, extrasList, customByClass = {}, {}, {}
		for _, r in ipairs(prof.items) do
			if r.Class and r.Item then
				blCache[r] = splitList(r.Blacklist)
				local cl, nm = r.Class:lower(), norm(r.Item)
				byClassName[cl] = byClassName[cl] or {}
				byClassName[cl][nm] = byClassName[cl][nm] or {}
				table.insert(byClassName[cl][nm], r)
			end
		end
		for _, key in ipairs(EXTRA_KEYS) do
			local r = prof.extras[key]
			if r then
				blCache[r] = splitList(r.Blacklist)
				table.insert(extrasList, { match = EXTRA_MATCH[key], rule = r })
			end
		end
		for _, r in ipairs(prof.custom) do
			if r.Class then
				blCache[r] = splitList(r.Blacklist)
				local cl = r.Class:lower()
				customByClass[cl] = customByClass[cl] or {}
				table.insert(customByClass[cl], r)
			end
		end
		index[pname] = { byClassName = byClassName, extrasList = extrasList, customByClass = customByClass }
	end
end
rebuild()

-- Named snapshots of every setting and rule. The running state (hop / AFK on or off) is never part
-- of a snapshot, so loading a config or auto loading after a server hop can't stop a running sniper.
local STORE_FILE = "MalemXSniper_configs.json"
local CAN_SAVE = type(writefile) == "function"
local store = { configs = {}, autoLoad = nil, active = nil, autoSave = false }

local function deepCopy(t) return HttpService:JSONDecode(HttpService:JSONEncode(t)) end

local function writeStore()
	if not CAN_SAVE then return end
	pcall(writefile, STORE_FILE, HttpService:JSONEncode(store))
end

local function loadStore()
	if type(isfile) ~= "function" or type(readfile) ~= "function" then return end
	pcall(function()
		if not isfile(STORE_FILE) then return end
		local data = HttpService:JSONDecode(readfile(STORE_FILE))
		if type(data) ~= "table" then return end
		if type(data.configs) == "table" then
			for name, c in pairs(data.configs) do
				if type(name) == "string" and type(c) == "table" then store.configs[name] = c end
			end
		end
		if type(data.autoLoad) == "string" then store.autoLoad = data.autoLoad end
		if type(data.active) == "string" then store.active = data.active end
		if type(data.autoSave) == "boolean" then store.autoSave = data.autoSave end
	end)
end

local function snapshot()
	local copy = deepCopy(cfg)
	copy.hopOn, copy.afkOn = nil, nil
	return copy
end

local function applyConfig(data)
	for k, v in pairs(data) do
		if k ~= "rules" and k ~= "hopOn" and k ~= "afkOn" and cfg[k] ~= nil and type(cfg[k]) == type(v) then cfg[k] = v end
	end
	if type(data.rules) == "table" then
		local hop, afk = newProfile(), newProfile()
		readProfile(hop, data.rules.hop)
		readProfile(afk, data.rules.afk)
		cfg.rules.hop, cfg.rules.afk = hop, afk
	end
	rebuild()
end

onConfigSaved = function()
	if store.autoSave and store.active and store.configs[store.active] then
		store.configs[store.active] = snapshot()
		writeStore()
	end
end

local function staticOk(r, c)
	if r.Enabled == false then return false end
	if r.MaxPrice == nil and r.Profit == nil then return false end -- never buy with no price limit
	if r.Tier ~= nil and r.Tier ~= c.tier then return false end
	if r.Shiny ~= nil and r.Shiny ~= c.sh then return false end
	local bl = blCache[r]
	if bl and (bl[c.name:lower()] or (c.id and bl[tostring(c.id):lower()])) then return false end
	return true
end

local function priceOk(r, c, cost)
	if r.MaxPrice and cost > r.MaxPrice then return false end
	if r.Profit then
		if not c.rap or c.rap <= 0 then return false end
		if (c.rap - cost) / c.rap * 100 < r.Profit then return false end
	end
	return true
end

-- static facts about a listing's item, cached once
local infoCache = setmetatable({}, { __mode = "k" })
local function getInfo(listing)
	local c = infoCache[listing]
	if c then return c end
	c = { ok = false, ver = {}, cands = {} }
	pcall(function()
		local item = listing.Item
		local d = item._data or {}
		c.name = tostring(item:GetName() or "?")
		c.class = tostring(item.Class and item.Class.Name or "?")
		c.id = d.id
		c.sh = d.sh == true
		c.tier = (c.class == "Pet") and (d.pt or 0) or (d.tn or 0)
		c.base = c.name
		if c.class ~= "Pet" then
			local base, n = splitTier(c.name)
			c.base = base
			if n and d.tn == nil then c.tier = n end
		end
		c.rap = tonumber(item:GetRAP()) or 0
		c.huge = item.IsHuge and item:IsHuge() or false
		c.titanic = item.IsTitanic and item:IsTitanic() or false
		c.garg = item.IsGargantuan and item:IsGargantuan() or false
		local okR, rar = pcall(function() return item:GetRarity()._id end)
		c.rarity = okR and rar or nil
		c.ok = true
	end)
	infoCache[listing] = c
	return c
end

local function candidates(c, pname)
	if c.ver[pname] == cfgVersion then return c.cands[pname] end
	local idx = index[pname]
	local list = {}
	local cl = c.class:lower()
	local byName = idx.byClassName[cl]
	if byName then
		local keys, seenKey = {}, {}
		for _, k in ipairs({ c.name, c.base, stripSize(c.base), c.id }) do
			k = norm(k)
			if not seenKey[k] then seenKey[k] = true; keys[#keys + 1] = k end
		end
		for _, k in ipairs(keys) do
			for _, r in ipairs(byName[k] or {}) do
				if staticOk(r, c) then list[#list + 1] = r end
			end
		end
	end
	for _, e in ipairs(idx.extrasList) do
		if e.match(c) and staticOk(e.rule, c) then list[#list + 1] = e.rule end
	end
	for _, r in ipairs(idx.customByClass[cl] or {}) do
		if staticOk(r, c) then list[#list + 1] = r end
	end
	c.ver[pname], c.cands[pname] = cfgVersion, list
	return list
end

local function gotoBooth(uid)
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local things = workspace:FindFirstChild("__THINGS")
	local booths = things and things:FindFirstChild("Booths")
	if not (root and booths) then return false end
	for _, b in ipairs(booths:GetChildren()) do
		if tostring(b:GetAttribute("Owner")) == tostring(uid) then
			local p = b:GetPivot()
			root.CFrame = CFrame.new(p.Position + p.LookVector * 5 + Vector3.new(0, 3, 0), p.Position)
			return true
		end
	end

end

local function sendWebhook(job)
	if not (cfg.webhookOn and cfg.webhookUrl ~= "" and requestFn) then return end
	pcall(requestFn, {
		Url = cfg.webhookUrl, Method = "POST", Headers = { ["Content-Type"] = "application/json" },
		Body = HttpService:JSONEncode({ username = "MalemX Sniper", embeds = { {
			title = "Sniped an item",
			description = string.format("**%s** x%d\nPaid `%s` each\nRAP `%s`\nProfit `%s`", job.name, job.qty,
				fmt(job.cost), fmt(job.rap), fmt((job.rap - job.cost) * job.qty)),
			color = 14033460,
		} } }),
	})
end

local function runJob(job)
	if not running() then return end
	busy = true
	stats.status = "Buying " .. job.name
	if cfg.teleport then
		gotoBooth(job.uid)
		if cfg.tpDelay > 0 then task.wait(cfg.tpDelay) end
	end
	while job.ready and workspace:GetServerTimeNow() < job.ready do RunService.Heartbeat:Wait() end
	local ok = false
	for _ = 1, math.max(cfg.retries, 1) do
		local s, res = pcall(function() return Purchase:InvokeServer(job.uid, { [job.key] = job.qty }, CALLER) end)
		if s and res then ok = true; break end
		RunService.Heartbeat:Wait()
	end
	lastActivity = os.clock()
	if ok then
		stats.bought += job.qty
		stats.spent += job.cost * job.qty
		stats.profit += (job.rap - job.cost) * job.qty
		stats.diamonds -= job.cost * job.qty
		stats.last = string.format("%s x%d @ %s", job.name, job.qty, fmt(job.cost))
		task.spawn(sendWebhook, job)
		handled[job.id] = os.clock() - 1.7 -- allow re-buying leftover stock quickly
	end
	busy = false
	stats.status = idleStatus()
end

task.spawn(function() -- one buy at a time, as fast as frames allow
	while alive do
		if #queue > 0 and not busy then
			runJob(table.remove(queue, 1))
		else
			RunService.Heartbeat:Wait()
		end
	end
end)

local function scan(pname)
	local seen, matched = 0, 0
	local me = LocalPlayer
	for _, tbl in pairs(boothTables()) do
		if type(tbl) == "table" then
			for _, b in pairs(tbl) do
				if type(b) == "table" and type(b.Listings) == "table" and b.Player and b.Player ~= me then
					local uid = b.Player.UserId
					for key, listing in pairs(b.Listings) do
						seen += 1
						local c = getInfo(listing)
						if c.ok then
							local cands = candidates(c, pname)
							if #cands > 0 then
								local cost = tonumber(listing.DiamondCost)
								if cost and cost > 0 then
									for _, r in ipairs(cands) do
										if priceOk(r, c, cost) then
											matched += 1
											local id = uid .. ":" .. tostring(key)
											local t0 = handled[id]
											if not t0 or os.clock() - t0 > 2 then
												local amount = (listing.Item._data and listing.Item._data._am) or 1
												local afford = diamondsKnown and math.floor((stats.diamonds - cfg.minGems) / cost) or math.huge
												local qty = math.min(amount, afford, cfg.maxPerBuy)
												if qty >= 1 then
													handled[id] = os.clock()
													table.insert(queue, { uid = uid, key = key, qty = qty, cost = cost, rap = c.rap,
														name = c.name, id = id, ready = listing.ReadyTimestamp })
													stats.note = nil
												else
													stats.note = string.format("Can't afford %s: you have %s gems, keeping %s", c.name, fmt(stats.diamonds), fmt(cfg.minGems))
												end
											end
											break
										end
									end
								end
							end
						end
					end
				end
			end
		end
	end
	stats.scanned, stats.matches = seen, matched
end

---------------------------------------------------------------- server hop (Hop Sniper only)
local hopping, failedServers = false, {}
local function hopServer()
	if hopping then return end
	hopping = true
	stats.status = "Finding a server..."
	task.spawn(function()
		local url = "https://games.roblox.com/v1/games/" .. PLAZA .. "/servers/Public?sortOrder=Desc&limit=100"
		local ok, body = pcall(function()
			if requestFn then
				local r = requestFn({ Url = url, Method = "GET" })
				return r and r.Body
			end
			return game:HttpGet(url)
		end)
		local pool = {}
		if ok and body then
			local okJ, data = pcall(function() return HttpService:JSONDecode(body) end)
			if okJ and type(data) == "table" and type(data.data) == "table" then
				for _, s in ipairs(data.data) do
					if s.playing and s.maxPlayers and s.playing < s.maxPlayers and s.id ~= game.JobId and not failedServers[s.id] then
						pool[#pool + 1] = s.id
					end
				end
			end
		end
		if #pool == 0 then
			pcall(TeleportService.Teleport, TeleportService, PLAZA, LocalPlayer)
		else
			local id = pool[math.random(1, #pool)]
			task.delay(4, function() failedServers[id] = true; hopping = false end) -- stuck? try another
			pcall(TeleportService.TeleportToPlaceInstance, TeleportService, PLAZA, id, LocalPlayer)
		end
	end)
end

-- closest booth (not yours) that has something you can afford; picks its cheapest listing
local function nearestBooth()
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return nil, "No character yet." end
	local best
	for _, tbl in pairs(boothTables()) do
		if type(tbl) == "table" then
			for _, b in pairs(tbl) do
				if type(b) == "table" and type(b.Listings) == "table" and b.Player and b.Player ~= LocalPlayer then
					local model = boothModel(b.Player.UserId)
					if model then
						local pickKey, pickCost, pickListing
						for key, listing in pairs(b.Listings) do
							local cost = tonumber(listing.DiamondCost)
							if cost and cost > 0 and (not diamondsKnown or cost <= stats.diamonds)
								and (not pickCost or cost < pickCost) then
								pickKey, pickCost, pickListing = key, cost, listing
							end
						end
						if pickKey then
							local dist = (model:GetPivot().Position - root.Position).Magnitude
							if not best or dist < best.dist then
								best = { uid = b.Player.UserId, owner = b.Player.Name, dist = dist, key = pickKey, cost = pickCost, listing = pickListing }
							end
						end
					end
				end
			end
		end
	end
	return best
end

local function testBuyNearest()
	if testBusy then return end
	testBusy = true
	task.spawn(function()
		local ok, err = pcall(function()
			if not Purchase or not getState then
				testSay("Not hooked yet. Join the Trading Plaza and wait for it to load.")
				return
			end
			local best, why = nearestBooth()
			if not best then
				testSay(why or "No booth with a listing you can afford was found.")
				return
			end
			local c = getInfo(best.listing)
			local name = c.ok and c.name or "an item"
			testSay(string.format("Nearest: %s's booth, %d studs away. %s for %s gems.", best.owner, math.floor(best.dist), name, fmt(best.cost)))
			if best.dist > BUY_RANGE then
				testSay("Out of range, teleporting...")
				if not gotoBooth(best.uid) then testSay("Teleport failed, trying the buy anyway.") end
				task.wait(math.max(cfg.tpDelay, 0.15))
			else
				testSay("In range, buying from here.")
			end
			local waited, readyAt = 0, best.listing.ReadyTimestamp
			while readyAt and workspace:GetServerTimeNow() < readyAt and waited < 10 do
				waited += RunService.Heartbeat:Wait()
			end
			for attempt = 1, math.max(cfg.retries, 1) do
				local s, res = pcall(function() return Purchase:InvokeServer(best.uid, { [best.key] = 1 }, CALLER) end)
				testSay(string.format("Attempt %d: call %s, server said %s", attempt, s and "went through" or "errored", tostring(res)))
				if s and res then
					stats.bought += 1
					stats.spent += best.cost
					stats.diamonds -= best.cost
					stats.last = string.format("%s x1 @ %s", name, fmt(best.cost))
					testSay("Bought it.")
					return
				end
				RunService.Heartbeat:Wait()
			end
			testSay("The server refused every attempt.")
		end)
		if not ok then testSay("Error: " .. tostring(err)) end
		testBusy = false
	end)
end

-- shows what the script actually sees on the booths, handy when a rule never matches
local function testListNearby()
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not (getState and root) then
		testSay("Not hooked yet. Join the Trading Plaza and wait for it to load.")

		for _, tbl in pairs(boothTables()) do

				for _, b in pairs(tbl) do
					if type(b) == "table" and type(b.Listings) == "table" and b.Player and b.Player ~= LocalPlayer then
						booths += 1
						local model = boothModel(b.Player.UserId)
						local dist = model and (model:GetPivot().Position - root.Position).Magnitude or math.huge
						for _, listing in pairs(b.Listings) do
							local c = getInfo(listing)

					end

	end)

	table.sort(rows, function(a, b) return a.dist < b.dist end)

	for i = 1, math.min(#rows, 8) do testSay(rows[i].text) end

local function clampOnScreen(obj)
	local vp = screenSize()
	local ap, as = obj.AbsolutePosition, obj.AbsoluteSize
	local dx, dy = 0, 0
	if ap.X < 4 then dx = 4 - ap.X elseif ap.X + as.X > vp.X - 4 then dx = vp.X - 4 - (ap.X + as.X) end
	if ap.Y < 4 then dy = 4 - ap.Y elseif ap.Y + as.Y > vp.Y - 4 then dy = vp.Y - 4 - (ap.Y + as.Y) end
	if dx ~= 0 or dy ~= 0 then obj.Position = obj.Position + UDim2.fromOffset(dx, dy) end
end

local function makeDraggable(handle, target, onTap)
	local dragging, moved, dragStart, startPos = false, false, nil, nil
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging, moved = true, false
			dragStart, startPos = input.Position, target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					if dragging and not moved and onTap then onTap() end
					dragging = false
				end
			end)
		end
	end)
	track(UIS.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local d = input.Position - dragStart
		if d.Magnitude > 4 then moved = true end
		if moved then
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
			clampOnScreen(target)
		end
	end))
end

		local maxPrice, profit = parseNum(priceBox.Text), tonumber(profitBox.Text)
		if not maxPrice and not profit then msg.Text = "Set a max price or a % below RAP."; return end

	strokeGrad.Rotation = (os.clock() * 40) % 360

	accDiamonds += dt
	if ready and accDiamonds > 0.25 then
		accDiamonds = 0
		local ok, v = pcall(function() return CurrencyCmds.Get("Diamonds") end)
		if ok and type(v) == "number" then stats.diamonds = v; diamondsKnown = true end
	end
	local pname = activeProfile()
	if ready and pname then
		acc += dt
		if cfg.scanInterval <= 0 or acc >= cfg.scanInterval then
			acc = 0
			local ok, err = pcall(scan, pname)
			if not ok then stats.status = "Scan error: " .. tostring(err) end
		end
	end

	accSlow += dt
	if accSlow > 0.5 then
		accSlow = 0
		local now = os.clock()
		stats.warn = (pname and ready and not hasActiveRules(pname))
			and ("No enabled price rules in the " .. (pname == "afk" and "AFK" or "Hop") .. " tab") or nil
		if cfg.hopOn and game.PlaceId ~= PLAZA and not hopping then
			hopping = true
			stats.status = "Going to the Trading Plaza..."
			pcall(TeleportService.Teleport, TeleportService, PLAZA, LocalPlayer)
		elseif cfg.hopOn and cfg.hopWhenIdle and ready and not hopping and #queue == 0 and not busy
			and now - math.max(lastActivity, joinClock) > cfg.hopDelay then
			hopServer()
		end
	end

	accUI += dt
	if accUI > 0.25 then
		accUI = 0
		headSub.Text = stats.status
		for _, update in ipairs(statViews) do update() end
	end
end))

track(TeleportService.TeleportInitFailed:Connect(function() hopping = false end))
