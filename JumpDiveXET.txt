-- Break All Zones (Halloween Games): breaks breakables in EVERY event zone, not just yours.
-- Uses the same game calls as VyHub (Breakables_JoinPetBulk / Breakables_PlayerDealDamage).

local genv = (type(getgenv) == "function" and getgenv()) or _G
if type(genv.BreakAllZones) == "table" and type(genv.BreakAllZones.Stop) == "function" then
	pcall(genv.BreakAllZones.Stop)
	task.wait(0.2)
end

local Players = game:GetService("Players")
local RS = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")
local UIS = game:GetService("UserInputService")
local player = Players.LocalPlayer

local INSTANCE_ID = "HalloweenGames"
local ZONE_COUNT = 8
local TAPS_PER_STEP = 12

local alive = true
local conns = {}
local breakOn, tapOn = false, false
local zoneOn = {}
for i = 1, ZONE_COUNT do zoneOn[i] = true end
local interval = 0.2 -- seconds between sweeps

local stats = { status = "Loading game modules...", targets = 0, pets = 0, broken = 0, taps = 0, perZone = {} }

---------------------------------------------------------------- game access (same approach as the hub)
local canSetId = type(setthreadidentity) == "function"
local execId = 7
if type(getthreadidentity) == "function" then
	local ok, id = pcall(getthreadidentity)
	if ok and type(id) == "number" then execId = id end
end

local function gcall(fn, ...)
	if type(fn) ~= "function" then return false, "not a function" end
	if canSetId then pcall(setthreadidentity, 2) end
	local r = table.pack(pcall(fn, ...))
	if canSetId then pcall(setthreadidentity, execId) end
	return table.unpack(r, 1, r.n)
end

local function req(path)
	local ok, mod = gcall(function()
		local node = RS
		for seg in string.gmatch(path, "[^%.]+") do
			node = node:FindFirstChild(seg)
			if not node then error("missing " .. path) end
		end
		return require(node)
	end)
	return ok and mod or nil
end

local Network, PlayerPet, BreakableFrontend
task.spawn(function()
	local t0 = os.clock()
	while alive and os.clock() - t0 < 60 do
		Network = Network or req("Library.Client.Network")
		PlayerPet = PlayerPet or req("Library.Client.PlayerPet")
		BreakableFrontend = BreakableFrontend or req("Library.Client.BreakableFrontend")
		if Network and PlayerPet then
			stats.status = "Ready"
			return
		end
		task.wait(1)
	end
	if alive and not (Network and PlayerPet) then stats.status = "Could not load game modules" end
end)

local function netFire(name, ...)
	if type(Network) ~= "table" or type(Network.Fire) ~= "function" then return false, "Network.Fire unavailable" end
	return gcall(Network.Fire, name, ...)
end
local function netUnreliable(name, ...)
	if type(Network) == "table" and type(Network.UnreliableFire) == "function" then
		return gcall(Network.UnreliableFire, name, ...)
	end
	return false
end

local function petEuids()
	if type(PlayerPet) ~= "table" or type(PlayerPet.GetAll) ~= "function" then return {} end
	local ok, out = gcall(function()
		local res, unfiltered = {}, {}
		local all = PlayerPet.GetAll()
		if type(all) ~= "table" then return res end
		local belongs = type(PlayerPet.BelongsToLocalPlayer) == "function" and PlayerPet.BelongsToLocalPlayer or nil
		for _, p in pairs(all) do
			if type(p) == "table" and p.euid ~= nil then
				unfiltered[#unfiltered + 1] = p.euid
				local keep = true
				if belongs then
					local okB, r = pcall(belongs, p)
					if okB and r == false then keep = false end
				end
				if keep then res[#res + 1] = p.euid end
			end
		end
		if #res == 0 then return unfiltered end
		return res
	end)
	if ok and type(out) == "table" then return out end
	return {}
end

---------------------------------------------------------------- breakables across ALL zones
local function collect()
	local list, perZone = {}, {}
	local things = workspace:FindFirstChild("__THINGS")
	local folder = things and things:FindFirstChild("Breakables")
	if not folder then return list, perZone end
	for _, m in ipairs(folder:GetChildren()) do
		if m:IsA("Model") and m:GetAttribute("ParentID") == INSTANCE_ID then
			local uid = m:GetAttribute("BreakableUID")
			if uid ~= nil then
				local z = tonumber(string.match(tostring(m:GetAttribute("AreaID")), "^Main_(%d+)$"))
				perZone[z or 0] = (perZone[z or 0] or 0) + 1
				if z == nil or zoneOn[z] then list[#list + 1] = { uid = uid, zone = z } end
			end
		end
	end
	return list, perZone
end

-- chests first, then weakest; zones are interleaved so every zone gets pets
local function prioritize(list)
	local ok, out = gcall(function()
		local byZone, zkeys = {}, {}
		local getFn = type(BreakableFrontend) == "table" and BreakableFrontend.Get or nil
		for _, b in ipairs(list) do
			local health, class = math.huge, "Normal"
			if getFn then
				local okG, info = pcall(getFn, b.uid)
				if okG and type(info) == "table" then
					if type(info.health) == "number" then health = info.health
					elseif type(info.maxHealth) == "number" then health = info.maxHealth end
					class = tostring(info.class or "Normal")
				end
			end
			if health > 0 then
				local z = b.zone or 0
				if not byZone[z] then byZone[z] = {}; zkeys[#zkeys + 1] = z end
				table.insert(byZone[z], { uid = b.uid, health = health, chest = class ~= "Normal" })
			end
		end
		table.sort(zkeys)
		for _, z in ipairs(zkeys) do
			table.sort(byZone[z], function(a, b)
				if a.chest ~= b.chest then return a.chest end
				if a.health ~= b.health then return a.health < b.health end
				return tostring(a.uid) < tostring(b.uid)
			end)
		end
		local res, i, more = {}, 1, true
		while more do
			more = false
			for _, z in ipairs(zkeys) do
				local e = byZone[z][i]
				if e then res[#res + 1] = e; more = true end
			end
			i += 1
		end
		return res
	end)
	return ok and out or {}
end

local lastAssign, lastSend, dirty, tapCursor = {}, 0, true, 0

local function step()
	if not (Network and PlayerPet) then return end
	local list, perZone = collect()
	stats.perZone = perZone
	if #list == 0 then
		stats.targets, stats.status = 0, "No breakables found. Be inside the event world."
		return
	end
	local ordered = prioritize(list)
	stats.targets = #ordered
	if #ordered == 0 then stats.status = "Nothing left to break"; return end

	if breakOn then
		local pets = petEuids()
		stats.pets = #pets
		if #pets == 0 then
			stats.status = "No pets found"
		else
			local map, changed = {}, dirty
			for i, e in ipairs(pets) do
				local target = ordered[((i - 1) % #ordered) + 1].uid
				map[e] = target
				if lastAssign[e] ~= target then changed = true end
			end
			lastAssign = map
			if changed or os.clock() - lastSend > 1 then
				lastSend, dirty = os.clock(), false
				local ok, err = netFire("Breakables_JoinPetBulk", map)
				if not ok then stats.status = "Pet send failed: " .. tostring(err); return end
			end
			stats.status = "Breaking " .. #ordered .. " targets"
		end
	end

	if tapOn then
		for _ = 1, math.min(#ordered, TAPS_PER_STEP) do
			tapCursor = tapCursor % #ordered + 1
			netUnreliable("Breakables_PlayerDealDamage", ordered[tapCursor].uid)
			stats.taps += 1
		end
		if not breakOn then stats.status = "Tapping " .. #ordered .. " targets" end
	end
end

local function track(c) conns[#conns + 1] = c; return c end

-- count what breaks and force a re-aim the moment one does
do
	local things = workspace:FindFirstChild("__THINGS")
	local folder = things and things:FindFirstChild("Breakables")
	if folder then
		track(folder.ChildRemoved:Connect(function(m)
			dirty = true
			if m:GetAttribute("ParentID") == INSTANCE_ID then stats.broken += 1 end
		end))
	end
end

local acc = 0
track(RunService.Heartbeat:Connect(function(dt)
	if not alive then return end
	acc += dt
	if acc < interval then return end
	acc = 0
	if breakOn or tapOn then
		local ok, err = pcall(step)
		if not ok then stats.status = "Error: " .. tostring(err) end
	else
		stats.status = "Off"
	end
end))

	local o = Instance.new(class)

-- minimize / close
local minimized = false

	minimized = not minimized

local function stop()
	alive = false
	for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	pcall(function() gui:Destroy() end)
	genv.BreakAllZones = nil
end
headBtn("x", -32, stop)
genv.BreakAllZones = { Stop = stop }

-- dragging (mouse + touch), from the header
local dragging, dragStart, startPos
head.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging, dragStart, startPos = true, input.Position, frame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)
track(UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local d = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
	end
end))
