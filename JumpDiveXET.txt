-- Broom Dodge + Air Jump + Noclip (draggable GUI)
local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")
local VIM = game:GetService("VirtualInputManager")
local Stats = game:GetService("Stats")

local player = Players.LocalPlayer
local dodgeOn, airJumpOn, noclipOn = false, false, false
local range = 10 -- studs, adjustable in the GUI
local rightOnly = true -- only react to brooms on your character's right side
local autoRange = true -- range grows with ping / low FPS (Range stepper = base range)

---------------------------------------------------------------- helpers
local function getHum()
	local char = player.Character
	return char and char:FindFirstChildOfClass("Humanoid")
end

local function isGrounded()
	local hum = getHum()
	if not hum then return false end
	local st = hum:GetState()
	return hum.FloorMaterial ~= Enum.Material.Air
		or st == Enum.HumanoidStateType.Running
		or st == Enum.HumanoidStateType.RunningNoPhysics
		or st == Enum.HumanoidStateType.Landed
end

local function doJump()
	local hum = getHum()
	if hum then
		hum.Jump = true
		hum:ChangeState(Enum.HumanoidStateType.Jumping)
	end
end

local function doDive()
	VIM:SendKeyEvent(true, Enum.KeyCode.T, false, game)
	task.wait(0.05)
	VIM:SendKeyEvent(false, Enum.KeyCode.T, false, game)
end

---------------------------------------------------------------- air jump (every 0.03s while airborne)
task.spawn(function()
	while true do
		task.wait(0.03)
		if airJumpOn and not isGrounded() then
			VIM:SendKeyEvent(true, Enum.KeyCode.Space, false, game) -- triggers the game's own double jump
			task.delay(0.01, function() VIM:SendKeyEvent(false, Enum.KeyCode.Space, false, game) end)
			local hum = getHum()
			if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
		end
	end
end)

---------------------------------------------------------------- noclip
local origCollide = {}
RunService.Stepped:Connect(function()
	if not noclipOn then return end
	local char = player.Character
	if not char then return end
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then
			if origCollide[d] == nil then origCollide[d] = d.CanCollide end
			d.CanCollide = false
		end
	end
end)

local function restoreCollide()
	for part, v in pairs(origCollide) do
		if part.Parent then part.CanCollide = v end
	end
	origCollide = {}
end

---------------------------------------------------------------- broom detection
local ARMS = { high = "HighBroomArm", low = "LowBroomArm" }
local armParts = { high = {}, low = {} }

local function getSlots()
	local node = workspace
	for _, n in ipairs({ "__THINGS", "__INSTANCE_CONTAINER", "Active", "HalloweenGames", "ArenaSlots" }) do
		node = node and node:FindFirstChild(n)
	end
	return node
end

local function scan()
	armParts = { high = {}, low = {} }
	local slots = getSlots()
	if not slots then return end
	for _, slot in ipairs(slots:GetChildren()) do
		local sweep = slot:FindFirstChild("BroomSweep")
		if sweep then
			for kind, name in pairs(ARMS) do
				local arm = sweep:FindFirstChild(name)
				if arm then
					if arm:IsA("BasePart") then table.insert(armParts[kind], arm) end
					for _, d in ipairs(arm:GetDescendants()) do
						if d:IsA("BasePart") then table.insert(armParts[kind], d) end
					end
				end
			end
		end
	end
end

-- distance from you to the arm's box, ignoring your own height; also returns the closest point
local function distToPart(p, pos)
	local flat = Vector3.new(pos.X, p.Position.Y, pos.Z)
	local lp = p.CFrame:PointToObjectSpace(flat)
	local h = p.Size / 2
	local c = Vector3.new(math.clamp(lp.X, -h.X, h.X), math.clamp(lp.Y, -h.Y, h.Y), math.clamp(lp.Z, -h.Z, h.Z))
	return (lp - c).Magnitude, p.CFrame:PointToWorldSpace(c)
end

local function nearest(kind, root)
	local pos = root.Position
	local best = math.huge
	for _, p in ipairs(armParts[kind]) do
		if p.Parent then
			local d, closest = distToPart(p, pos)
			if not rightOnly or root.CFrame:PointToObjectSpace(closest).X >= -0.5 then
				if d < best then best = d end
			end
		end
	end
	return best
end

	local b = Instance.new("TextButton")

		local on = getOn()
		b.Text = textFn(on)
		TweenService:Create(b, info, {
			BackgroundColor3 = on and green or Color3.fromRGB(45, 45, 56),
			TextColor3 = on and Color3.fromRGB(20, 20, 25) or dim,
		}):Play()
	end
	b.MouseButton1Click:Connect(function() flip(); refresh() end)
	refresh()
end

-- dragging (mouse + touch)
local dragging, dragStart, startPos
frame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging, dragStart, startPos = true, input.Position, frame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)
UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local d = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
	end
end)

---------------------------------------------------------------- auto dodge loop
local LOCKOUT = 0.8 -- min seconds between triggers of the same broom
local APPROACH = -0.1 -- studs/sec; broom must be closing in faster than this
local state = {
	high = { armed = true, last = 0, prev = math.huge, rate = 0, speed = 0 },
	low = { armed = true, last = 0, prev = math.huge, rate = 0, speed = 0 },
}
local lastScan, lastUI, lastPing = 0, 0, 0
local pingMs, dtAvg = 0, 1 / 60

-- trigger distance for a broom: base range + how far it travels during your total delay
-- (round-trip ping + ~2 frames), so higher ping / lower FPS makes it dodge earlier
local function effRange(s, latency)
	if not autoRange then return range end
	return math.clamp(range + s.speed * latency, 4, 30)
end

RunService.Heartbeat:Connect(function(dt)
	local now = os.clock()
	if now - lastScan > 1 then lastScan = now; scan() end

	dtAvg = dtAvg * 0.9 + dt * 0.1
	if now - lastPing > 0.5 then
		lastPing = now
		local ok, v = pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() end)
		if ok and v then pingMs = pingMs == 0 and v or pingMs * 0.7 + v * 0.3 end
	end
	local latency = pingMs / 1000 + dtAvg * 2

	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end

	local dHigh, dLow = nearest("high", root), nearest("low", root)

	if now - lastUI > 0.1 then
		lastUI = now
		local function f(d) return d == math.huge and "-" or string.format("%.1f", d) end
		local sl = state.low
		readout.Text = "H " .. f(dHigh) .. " | L " .. f(dLow) .. (sl.armed and "  armed" or "  locked")
			.. string.format("\nRng %.1f | %dms | %d FPS", effRange(sl, latency), math.floor(pingMs), math.floor(1 / dtAvg))
	end
	if not dodgeOn then return end

	-- once per sweep, only for a broom moving toward you
	local function update(kind, d, action, needsGround)
		local s = state[kind]
		if d == math.huge then s.prev, s.rate, s.armed = math.huge, 0, true; return end
		if s.prev ~= math.huge and dt > 0 then
			s.rate = s.rate * 0.6 + ((d - s.prev) / dt) * 0.4
		end
		s.prev = d
		s.speed = s.speed * 0.9 + math.max(-s.rate, 0) * 0.1 -- smoothed approach speed (studs/sec)
		local r = effRange(s, latency)
		if s.armed then
			if d <= r and s.rate < APPROACH and now - s.last > LOCKOUT
				and (not needsGround or isGrounded()) then
				s.armed, s.last = false, now
				action()
			end
		elseif d > r + 2 then
			s.armed = true
		end
	end

	update("high", dHigh, function() task.spawn(doDive) end, false)
	update("low", dLow, doJump, true)
end)
